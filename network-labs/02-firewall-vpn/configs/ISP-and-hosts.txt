! ============================================================
! ISP  (Cisco 2911) — "the internet" in the middle
! All its networks are directly connected, so no routes needed.
! It knows NOTHING about 172.16.x.x — private IPs never cross the internet.
! ============================================================
enable
configure terminal
hostname ISP
no ip domain-lookup
interface GigabitEthernet0/0
 description To HQ-FW
 ip address 203.0.113.1 255.255.255.252
 no shutdown
interface GigabitEthernet0/1
 description To BRANCH-FW
 ip address 198.51.100.1 255.255.255.252
 no shutdown
interface GigabitEthernet0/2
 description Internet user / attacker
 ip address 192.0.2.1 255.255.255.0
 no shutdown
end
write memory
